1 Executive Summary

For a self-funded single citizen aged 35 with $100,000 available CPF/cash, the path that maximises projected 5-year net worth is a 3-Room resale flat in Punggol (~$192,736) versus $179,914 for the next-best alternative (Punggol BTO) under central interest rates (~3.5%) and central price growth. This report combines three HDB transaction datasets and 2019 Master Plan geo-spatial accessibility data into a multi-stage analytical framework (Modules 1–4) culminating in an interactive decision engine.

6 The Net-Worth Simulator Engine

Code 
rent_growth_annual <- 0.035
bto_growth_annual <- 0.021

monthly_payment <- function(principal, annual_rate, years) {
  r <- annual_rate / 12
  n <- years * 12
  if (r == 0) principal / n else principal * r / (1 - (1 + r)^(-n))
}

simulate_networth_v3 <- function(path, price, annual_rate, growth, rules,
                                  cash_injection = NULL, base_rent = 2200,
                                  rent_growth = rent_growth_annual) {
  h <- rules$horizon_years

  if (path == "rent") {
    yearly_rent_outlays <- numeric(h)
    for (year in 1:h) {
      yearly_rent_outlays[year] <- (base_rent * 12) * (1 + rent_growth)^(year - 1)
    }
    total_rent_paid <- sum(yearly_rent_outlays)
    return(tibble(
      net_worth_5y = -total_rent_paid,
      monthly_housing = base_rent
    ))
  }

  min_downpayment <- price * (1 - rules$ltv)
  down <- if (is.null(cash_injection) || cash_injection < min_downpayment) min_downpayment else cash_injection
  loan <- max(0, price - down)
  pmt <- monthly_payment(loan, annual_rate, rules$loan_years)
  paid <- pmt * 12 * h

  r <- annual_rate / 12
  n <- rules$loan_years * 12
  k <- h * 12
  bal <- if (loan == 0) 0 else loan * ((1 + r)^n - (1 + r)^k) / ((1 + r)^n - 1)

  value_5y <- price * (1 + growth)^h
  equity <- value_5y - bal

  return(tibble(
    net_worth_5y = equity - paid - down,
    monthly_housing = pmt
  ))
}
